# Security Quick Reference - Developer Guide\n\nQuick checklists and snippets for secure coding in TechTrek.\n\n---\n\n## Authentication\n\n### ✅ DO: Secure Login\n```javascript\nexport async function POST(req) {\n  const { email, password } = req.body;\n  \n  // 1. Validate input\n  if (!email || !password) throw new Error('Missing fields');\n  \n  // 2. Sanitize email\n  const cleanEmail = sanitizeEmail(email);\n  \n  // 3. Never log password\n  console.log(`Login for: ${cleanEmail}`);\n  \n  // 4. Check rate limit\n  if (await isRateLimited(cleanEmail)) {\n    throw new Error('Too many attempts');\n  }\n  \n  // 5. Find user\n  const user = await User.findOne({ email: cleanEmail }).select('+passwordHash');\n  if (!user) throw new Error('Invalid email or password');\n  \n  // 6. Verify with bcrypt\n  const match = await verifyPassword(password, user.passwordHash);\n  if (!match) throw new Error('Invalid email or password');\n  \n  // 7. Generate secure token\n  const token = generateAccessToken(user._id);\n  \n  // 8. Set secure cookie\n  response.cookies.set('refreshToken', token, {\n    httpOnly: true,\n    secure: true,\n    sameSite: 'strict'\n  });\n  \n  return response;\n}\n```\n\n### ❌ DON'T: Insecure Login\n```javascript\n// WRONG - stores plaintext password\nconst user = await User.findOne({ email });\nif (user.password === password) {}\n\n// WRONG - logs sensitive data\nconsole.log('Login attempt:', req.body);\n\n// WRONG - no rate limiting\nfor (i = 0; i < 1000; i++) attemptLogin();\n\n// WRONG - generic token\nconst token = user.id + Date.now();\n\n// WRONG - no secure cookie flags\nresponse.cookies.set('token', token);\n```\n\n---\n\n## Input Validation\n\n### ✅ DO: Validate & Sanitize\n```javascript\nexport async function POST(req) {\n  let { email, courseId, title } = req.body;\n  \n  // Validate email format\n  email = sanitizeEmail(email);\n  \n  // Validate MongoDB ObjectId\n  courseId = validateObjectId(courseId);\n  \n  // Sanitize string input\n  title = sanitizeString(title, {\n    minLength: 1,\n    maxLength: 200,\n    allowSpecialChars: true\n  });\n  \n  // Now safe to use\n  const course = await Course.findById(courseId);\n  return Response.json(course);\n}\n```\n\n### ❌ DON'T: Use Raw Input\n```javascript\n// WRONG - NoSQL injection vulnerability\nconst course = await Course.findOne({ _id: req.body.courseId });\n// Attack: ?courseId={$ne: null} returns all courses\n\n// WRONG - XSS vulnerability\nreturn Response.json({ title: req.body.title }); // Could contain <script>\n\n// WRONG - No validation\nif (req.body.email.includes('@')) {\n  // Email could be anything\n}\n```\n\n---\n\n## API Security\n\n### ✅ DO: Secure API Route\n```javascript\nimport { verifyAccessToken } from '@/lib/auth';\nimport { sanitizeInput } from '@/lib/sanitizer';\nimport { corsMiddleware } from '@/lib/cors';\n\nexport async function POST(req) {\n  try {\n    // 1. Check CORS\n    if (corsMiddleware(req)) return;\n    \n    // 2. Verify authentication\n    const authHeader = req.headers.get('Authorization');\n    const user = verifyAccessToken(authHeader);\n    \n    // 3. Sanitize input\n    let data = await req.json();\n    data = sanitizeInput(data);\n    \n    // 4. Validate business logic\n    if (user.id !== data.userId && user.role !== 'admin') {\n      return Response.json(\n        { error: 'Unauthorized' },\n        { status: 403 }\n      );\n    }\n    \n    // 5. Perform operation\n    const result = await Course.create(data);\n    \n    // 6. Log audit\n    await logAudit(user.id, 'CREATE_COURSE', 'course', req.ip);\n    \n    return Response.json(result);\n  } catch (err) {\n    // 7. Generic error response\n    console.error('API error:', err.message);\n    return Response.json(\n      { error: 'Internal server error' },\n      { status: 500 }\n    );\n  }\n}\n```\n\n### ❌ DON'T: Insecure API\n```javascript\n// WRONG - no authentication check\nexport async function POST(req) {\n  const data = await req.json();\n  return Response.json(await Course.create(data));\n}\n\n// WRONG - leaks error details\ncatch (err) {\n  return Response.json({ error: err.message }, { status: 500 });\n  // Attacker sees: \"Invalid MongoDB ObjectId\"\n}\n\n// WRONG - returns sensitive data\nconst user = await User.findById(userId);\nreturn Response.json(user); // Includes passwordHash!\n```\n\n---\n\n## Password Security\n\n### ✅ DO: Strong Password\n```javascript\nimport { validatePasswordStrength } from '@/lib/auth';\n\nexport async function POST(req) {\n  const { password } = req.body;\n  \n  try {\n    // Validate strength\n    validatePasswordStrength(password);\n    // Requires: 8+ chars, uppercase, lowercase, number, special char\n    \n    // Hash with bcrypt 12 rounds\n    const hash = await hashPassword(password);\n    \n    await User.create({ ..., passwordHash: hash });\n  } catch (err) {\n    return Response.json({ error: err.message }, { status: 400 });\n  }\n}\n```\n\n### ❌ DON'T: Weak Password\n```javascript\n// WRONG - only checks length\nif (password.length < 8) throw new Error('Too short');\n\n// WRONG - stores plaintext\nawait User.create({ password }); // NEVER!\n\n// WRONG - weak bcrypt\nconst hash = await bcrypt.hash(password, 6); // Only 6 rounds!\n```\n\n---\n\n## Database Queries\n\n### ✅ DO: Safe Query\n```javascript\nimport { validateObjectId } from '@/lib/sanitizer';\nimport { Types } from 'mongoose';\n\nexport async function GET(req) {\n  try {\n    // Validate ObjectId format\n    const courseId = validateObjectId(req.query.id);\n    \n    // Cast to MongoDB ObjectId type\n    const course = await Course.findById(\n      new Types.ObjectId(courseId)\n    );\n    \n    return Response.json(course);\n  } catch (err) {\n    return Response.json({ error: 'Invalid ID' }, { status: 400 });\n  }\n}\n```\n\n### ❌ DON'T: Dangerous Query\n```javascript\n// WRONG - direct query with unsanitized input\nconst course = await Course.findOne({ _id: req.query.id });\n// Attack: ?id={$ne: null} returns any course\n\n// WRONG - operator injection\nconst users = await User.find(req.query.filter);\n// Attack: ?filter={role: {$ne: 'user'}} bypasses authorization\n```\n\n---\n\n## Authentication Tokens\n\n### ✅ DO: Secure Tokens\n```javascript\nimport jwt from 'jsonwebtoken';\n\n// Generate access token (short-lived, 15 min)\nexport function generateAccessToken(userId) {\n  return jwt.sign(\n    { userId, type: 'access' },\n    process.env.JWT_ACCESS_SECRET,\n    {\n      algorithm: 'HS256',\n      issuer: 'techtrek',\n      audience: 'techtrek-users',\n      expiresIn: '15m'\n    }\n  );\n}\n\n// Generate refresh token (long-lived, 7 days)\nexport function generateRefreshToken(userId) {\n  return jwt.sign(\n    { userId, type: 'refresh' },\n    process.env.JWT_REFRESH_SECRET,\n    {\n      algorithm: 'HS256',\n      issuer: 'techtrek',\n      audience: 'techtrek-users',\n      expiresIn: '7d'\n    }\n  );\n}\n\n// Verify token\nexport function verifyAccessToken(token) {\n  try {\n    const decoded = jwt.verify(\n      token,\n      process.env.JWT_ACCESS_SECRET,\n      {\n        algorithms: ['HS256'],\n        issuer: 'techtrek',\n        audience: 'techtrek-users'\n      }\n    );\n    if (decoded.type !== 'access') throw new Error('Invalid token type');\n    return decoded;\n  } catch (err) {\n    if (err.name === 'TokenExpiredError') throw new Error('Token expired');\n    throw new Error('Invalid token');\n  }\n}\n\n// Store refresh token cookie\nresponse.cookies.set('refreshToken', token, {\n  httpOnly: true,    // Not accessible from JavaScript (prevents XSS theft)\n  secure: true,      // Only sent over HTTPS\n  sameSite: 'strict', // Prevents CSRF\n  maxAge: 7 * 24 * 60 * 60 * 1000 // 7 days\n});\n```\n\n### ❌ DON'T: Insecure Tokens\n```javascript\n// WRONG - no expiry\njwt.sign({ userId }, secret); // Valid forever!\n\n// WRONG - no issuer/audience check\njwt.verify(token, secret);\n\n// WRONG - leaky cookie\nresponse.cookies.set('token', token); // Accessible from JavaScript!\n\n// WRONG - refresh token not rotated\n// After 7 days, same token still works even if compromised\n```\n\n---\n\n## Logging\n\n### ✅ DO: Secure Logging\n```javascript\n// Log only non-sensitive data\nconsole.log(`User login: ${email}`);\nconsole.log(`Course created: ${courseId}`);\n\n// Log important events for audit\nawait AuditLog.create({\n  userId: user._id,\n  action: 'LOGIN',\n  ipAddress: req.ip,\n  timestamp: new Date()\n});\n\n// Log security events\nawait AuditLog.create({\n  userId: user._id,\n  action: 'ACCESS_DENIED',\n  reason: 'Unauthorized resource access',\n  ipAddress: req.ip\n});\n```\n\n### ❌ DON'T: Leak Secrets in Logs\n```javascript\n// WRONG - logs password\nconsole.log('Login attempt:', req.body);\n\n// WRONG - logs database URI\nconsole.log('DB connection:', process.env.MONGODB_URI);\n\n// WRONG - logs tokens\nconsole.log('Authorization:', req.headers.authorization);\n\n// WRONG - logs raw errors with schema info\ncatch (err) {\n  console.error(err); // Might expose DB structure\n}\n```\n\n---\n\n## Rate Limiting\n\n### ✅ DO: Rate Limit Critical Endpoints\n```javascript\nimport rateLimit from 'express-rate-limit';\n\nconst authLimiter = rateLimit({\n  windowMs: 15 * 60 * 1000, // 15 minutes\n  max: 5, // 5 attempts per window\n  message: 'Too many authentication attempts',\n  keyGenerator: (req) => req.body?.email || req.ip\n});\n\nexport async function POST(req) {\n  // Apply limiter\n  // In production, use middleware\n  if (await isRateLimited(req.body.email)) {\n    return Response.json(\n      { error: 'Too many attempts' },\n      { status: 429 }\n    );\n  }\n  // ... continue with login\n}\n```\n\n### ❌ DON'T: Allow Unlimited Attempts\n```javascript\n// WRONG - no rate limiting\nfor (i = 0; i < 10000; i++) {\n  login(email, 'password123'); // Attacker can brute-force\n}\n\n// WRONG - rate limit too high\nrateLimit({ max: 1000 }); // Allows too many attempts\n\n// WRONG - rate limit by IP only\nrateLimit({ keyGenerator: (req) => req.ip });\n// Multiple emails from same IP could still attack\n```\n\n---\n\n## Error Handling\n\n### ✅ DO: Generic Errors\n```javascript\nexport async function POST(req) {\n  try {\n    const user = await User.findOne({ email });\n    if (!user) {\n      // Generic error (don't reveal if email exists)\n      return Response.json(\n        { error: 'Invalid email or password' },\n        { status: 401 }\n      );\n    }\n  } catch (err) {\n    // Generic error in production\n    console.error('Error:', err.message); // Log internally\n    return Response.json(\n      { error: process.env.NODE_ENV === 'production' \n        ? 'Internal server error' \n        : err.message },\n      { status: 500 }\n    );\n  }\n}\n```\n\n### ❌ DON'T: Leak Information in Errors\n```javascript\n// WRONG - reveals email enumeration\nreturn Response.json({ error: 'Email not found' }, { status: 404 });\n\n// WRONG - leaks database schema\ncatch (err) {\n  return Response.json({ error: err.message }, { status: 500 });\n  // Error: \"Cast to ObjectId failed for value 'abc'\" reveals schema\n}\n\n// WRONG - SQL-like injection error\n// Error: \"SyntaxError in query\" reveals query structure\n```\n\n---\n\n## CORS & Headers\n\n### ✅ DO: Secure CORS\n```javascript\n// next.config.js\nasync headers() {\n  return [{\n    source: '/:path*',\n    headers: [\n      {\n        key: 'Access-Control-Allow-Origin',\n        value: process.env.FRONTEND_URL // Only allow your frontend\n      },\n      {\n        key: 'Access-Control-Allow-Methods',\n        value: 'GET,POST,PUT,DELETE,OPTIONS'\n      },\n      {\n        key: 'Access-Control-Allow-Headers',\n        value: 'Content-Type,Authorization'\n      },\n      {\n        key: 'Access-Control-Allow-Credentials',\n        value: 'true'\n      },\n      {\n        key: 'Strict-Transport-Security',\n        value: 'max-age=31536000; includeSubDomains'\n      },\n      {\n        key: 'X-Content-Type-Options',\n        value: 'nosniff'\n      },\n      {\n        key: 'X-Frame-Options',\n        value: 'DENY'\n      }\n    ]\n  }];\n}\n```\n\n### ❌ DON'T: Allow All Origins\n```javascript\n// WRONG - allows any origin\nAccess-Control-Allow-Origin: *\n\n// WRONG - allows wildcard with credentials\nAccess-Control-Allow-Origin: *\nAccess-Control-Allow-Credentials: true // Contradiction!\n\n// WRONG - no security headers\n// Missing HSTS, CSP, X-Frame-Options, etc.\n```\n\n---\n\n## Checklist: Before Pushing Code\n\n- [ ] All passwords hashed with bcrypt 12+ rounds\n- [ ] No plaintext passwords in code\n- [ ] No sensitive data in logs\n- [ ] All user input validated and sanitized\n- [ ] All database queries validated (especially ObjectIds)\n- [ ] JWT tokens have expiry, issuer, audience, algorithm\n- [ ] Refresh tokens in httpOnly secure cookies\n- [ ] Rate limiting on auth endpoints\n- [ ] CORS restricted to FRONTEND_URL\n- [ ] Security headers set\n- [ ] Error messages generic in production\n- [ ] No secrets in version control\n- [ ] Audit logging for sensitive operations\n- [ ] Authorization checks on protected routes\n- [ ] CSRF tokens on state-changing requests\n\n---\n\n## Quick Commands\n\n```bash\n# Generate random secrets\nnode -e \"console.log(require('crypto').randomBytes(32).toString('hex'))\"\n\n# Check for plaintext secrets\ngit grep -i password -- '*.js' '*.ts'\n\n# Find console.log in auth files\ngrep -r \"console\\.log\" app/api/auth/\n\n# Check bcrypt rounds\ngrep -r \"bcrypt\\.hash\" lib/\n\n# List all API routes\nfind app/api -name \"route.js\" -o -name \"route.ts\"\n```\n\n---\n\n**Remember:** Security is not a feature, it's a requirement.
